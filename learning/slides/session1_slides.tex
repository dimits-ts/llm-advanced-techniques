    \begin{frame}{Why finetune LLMs?}
    	
        \begin{itemize}
            \item \textbf{Prompting:} Changes \alert{input behavior}. We guide
            the model with instructions and examples. \item
            \textbf{Fine-Tuning:} Changes \alert{model parameters}. We switch
            the \alert{knowledge} of the model.
        \end{itemize}

        \begin{block}{Reminder}
            \textbf{Fine-Tuning (FT):} The process of further training a
            pre-trained model on a smaller, domain-specific dataset to adapt its
            general knowledge to a specialized task.
        \end{block}
    \end{frame}
    
    \begin{frame}{Pretraining vs Finetuning}
    	\begin{block}{Reminder}
    		%Link with Katerinas lesson
    		\begin{itemize}
    			\item \alert{Pre-training} provides the model with initial, \alert{general} knowledge by utilizing \alert{any} and all data sources. 
    			\item \alert{Finetuning} shifts existing knowledge, to \alert{specific} domains using \alert{domain-specific} datasets.
    		\end{itemize}
    	\end{block}
    \end{frame}

    \subsection{Why not (fully) finetune LLMs?}
	
	\begin{frame}

    \begin{frame}{High Resource Cost}
        \begin{itemize}
                \item \textbf{Computational Power:} Full FT requires
                \alert{massive hardware requirements} and
                \alert{thousands of compute hours} for large models (e.g.,
                Llama-70B).

                \item \textbf{Storage:} Each specialized task requires a
                separate, full copy of the massive foundational model.
        \end{itemize}
    \end{frame}

    \begin{frame}{Operational Overhead}
        \begin{itemize}
          
                \item Training is \alert{slow}. \item Deployment becomes
                complex, managing \alert{multiple large models} for different
                applications.
                \begin{itemize}
                    \item A single 13B model may require 26GB of storage. \item
                    10 finetuned models may thus require 260GB.
                \end{itemize}
                \item High risk of \alert{Catastrophic Forgetting}---the model
                forgets its general knowledge while learning the specific task.
        \end{itemize}
    \end{frame}

    \begin{frame}{Scalability issues}
        \begin{columns}
            \begin{column}{0.45\textwidth}
                \begin{block}{BERT}
                    \centering
                    \begin{itemize}
                        \item \textbf{Model Size:} $\sim$340M Parameters \item
                        \textbf{VRAM Requirement:} $\approx$ 8-12GB \item
                        \textbf{Training Time:} Minutes to a few hours
                    \end{itemize}
                \end{block}
            \end{column}

            \begin{column}{0.45\textwidth}
                \begin{block}{Llama 70B}
                    \centering
                    \begin{itemize}
                        \item \textbf{Model Size:} $\sim$70 Billion Parameters
                        \item \textbf{VRAM Requirement:} $\approx$1000GB+ \item
                        \textbf{Training Time:} Days to Weeks (even with
                        specialized hardware)
                    \end{itemize}
                \end{block}
            \end{column}
        \end{columns}

        \begin{center}
            LLMs are \alert{too large} to run full-finetuning. But do we need
            to?
        \end{center}
    \end{frame}

    \begin{frame}{Parameter-Efficient Fine-Tuning (PEFT)}
    	\begin{definition}
    		\textbf{PEFT}:  A set of techniques used to adapt large pre-trained LLMs to specific tasks by updating only a \alert{small subset of their parameters}, rather than retraining the entire model.
    	\end{definition}
    	\vspace{2em}
    	\begin{itemize}
    		\item Many algorithms developed through the years (adapters, prompt/prefix tuning etc.).
    		\item However, the \alert{LoRA}  family of methods has dominated the LLM space in the last few years.
    	\end{itemize}
    \end{frame}

    \begin{frame}{PEFT: LoRA}
        \begin{itemize}
            \item Low-Rank (LoRA) Matrix Decomposition is a specific PEFT
            algorithm that exploits the fact that weight updates in LLMs have a
            \alert{``low intrinsic rank''}.

            \item Low rank matrices enable us to \alert{scalably} represent
            information.
        \end{itemize}
    \end{frame}

    \begin{frame}{Low-rank matrices}
        \begin{block}{What is a matrix?}
            A matrix is essentially a organized collection of data (rows and
            columns). For large datasets, these matrices can become incredibly
            complex, requiring massive storage and computational power.
        \end{block}

        \begin{exampleblock}{Example: Data matrix}
            \[
            \begin{bmatrix}
                2 & 4 & 6 & 8 & 10\\
                1 & 2 & 3 & 4 & 5\\
                3 & 6 & 9 & 12 & 15\\
                4 & 8 & 12 & 16 & 20
            \end{bmatrix}
            \]
        \end{exampleblock}
    \end{frame}

    \begin{frame}{Low-rank matrices}
        \begin{block}{Matrix rank}
            \begin{itemize}
                \item Rank $r$ represents the number of genuinely independent columns (or rows!) within the matrix $M$.\note{Can't be inferred by the rest of the information.} \item If $M \in \mathbb{R}^{m\times n}$, then $rank(M) \leq min(m, n)$.
            \end{itemize}
        \end{block}

       	\begin{exampleblock}{Example: Music}
	       	\begin{itemize}
	       		\item Imagine a piano with 88 keys.
	       		\item Most songs don't require all 88 keys; they can be described by melody, rhythm etc.
	       		\item Low-rank updates capture changing \alert{general patterns}.
	       	\end{itemize}
       \end{exampleblock}
    \end{frame}

    \begin{frame}{Low-rank matrices}
        \begin{block}{Rank Factorization}
            If a matrix $M$ has a low rank $r$, we can approximate it (or
            exactly represent it) by factoring it into a product of two simpler
            matrices.
        \end{block}

        \begin{exampleblock}{Example: Rank 1 matrix}
            \[
            \begin{bmatrix}
                2 & 4 & 6 & 8 & 10\\
                1 & 2 & 3 & 4 & 5\\
                3 & 6 & 9 & 12 & 15\\
                4 & 8 & 12 & 16 & 20
            \end{bmatrix}
            =
            \begin{bmatrix}
                2\\
                1\\
                3\\
                4
            \end{bmatrix}
            \begin{bmatrix}
                1 & 2 & 3 & 4 & 5
            \end{bmatrix}.
            \]
        \end{exampleblock}
    \end{frame}
    
    \begin{frame}{Why are LLM updates typically low-rank?}
    	\note{This is a hypothesis of the seminal paper. These explanations originate from prior work of that paper, verified by other studies.}
    	
    	\begin{block}{Many possible explanations}
    		\begin{itemize}
    			\item Updates generally follow the direction of existing knowledge $\rightarrow$ the updates may often just be \alert{re-weighting} of existing weights.
    			\item Underlying \alert{ML problems} are usually inherently low-rank.
    		\end{itemize}
    	\end{block}
    \end{frame}
    
    \begin{frame}{Why are LLM updates typically low-rank?}
    	\begin{columns}
    		\begin{column}{0.45\columnwidth}
    			\begin{figure}
    				\includegraphics[width=\columnwidth]{embeddings.png}
    				\caption{\url{https://aclanthology.org/2025.emnlp-main.1410.pdf}}
    			\end{figure}
    		\end{column}
    		
    		\begin{column}{0.45\columnwidth}
    			\begin{block}{Large embedding spaces}
    					Modern NLP models are \alert{over-parameterized} $\rightarrow$ not enough information to fill in higher ranks of weights.
    			\end{block}
    		\end{column}
    	\end{columns}
    \end{frame}

    \begin{frame}{LoRA: Low-rank decomposition}
        \[
        \underbrace{
            A_{m \times n}
        }_{\text{large matrix}}
        \approx
        \underbrace{
            U_{m \times r}
        }_{\text{left factors}}
        \underbrace{
            V_{r \times n}
        }_{\text{right factors}},
        \qquad r \ll \min(m,n).
        \]

        \[
        A_{10000 \times 5000}
        \approx
        U_{10000 \times 50}
        V_{50 \times 5000},
        \]

        \[
        \underbrace{10000 \times 5000}_{50,000,000\ \text{parameters}}
        \;\longrightarrow\;
        \underbrace{10000 \times 50 + 50 \times 5000}_{750,000\ \text{parameters}}.
        \]
    \end{frame}

    \begin{frame}{LoRA: Mathematical Foundation}
    	\begin{columns}
    		\begin{column}{0.5\textwidth}
    			 \begin{figure}
    				\includegraphics[height=0.7\textheight]{lora1.drawio.png} \centering
    			\end{figure}
    		\end{column}
    		
    		\begin{column}{0.5\textwidth}
    			\begin{block}{Full SFT}
    					\begin{itemize}
    					\item $W \in \mathbb{R}^{d \times k}$ weights. \item $X$ input,
    					$h$ output.  
    					\item Fwd pass: $h = WX$.
    					\item Bckwd pass: $W = W + \alpha \nabla h $
    					%\item $B  \in \mathbb{R}^{d \times r} \rightarrow$ zeros. \item	$A \in \mathbb{R}^{r \times k} \rightarrow$ Gaussian. \item $r \ll \min(d, k)$, hence $A, B$ are much smaller than $W$.
    				\end{itemize}
    			\end{block}
    		\end{column}
    	\end{columns}
	\end{frame}
	
	\begin{frame}{LoRA: Mathematical Foundation}
		\begin{columns}
			\begin{column}{0.5\textwidth}
				\begin{figure}
					\includegraphics[height=0.7\textheight]{lora2.drawio.png} \centering
				\end{figure}
			\end{column}
			
			\begin{column}{0.5\textwidth}
				\begin{block}{LoRA}
					\begin{itemize}
						\item	$A \in \mathbb{R}^{r \times k} \rightarrow$ Gaussian.
						\item $B  \in \mathbb{R}^{d \times r} \rightarrow$ zeros.  \item $r \ll \min(d, k)$, hence $A, B$ are much smaller than $W$.
					\end{itemize}
				\end{block}
				
				\begin{exampleblock}{Example: Music}
					$A$ could represent the genre of music, while $B$ the melody, rhythm etc.
				\end{exampleblock}
			\end{column}
		\end{columns}
	\end{frame}
    
    \begin{frame}{LoRA: Mathematical Foundation}
    	\begin{columns}
    		\begin{column}{0.5\textwidth}
    			\begin{figure}
    				\includegraphics[height=0.7\textheight]{lora3.drawio.png} \centering
    			\end{figure}
    		\end{column}
    		
    		\begin{column}{0.5\textwidth}
    			\begin{block}{LoRA}
    				\begin{itemize}
    					\item During inference, we compute $H = W X + \Delta W X$.
    					\item Alternatively, merge weights: $W' = W + \frac{\alpha}{r} (A B)$
    				\end{itemize}
    			\end{block}
    		\end{column}
    	\end{columns}
    \end{frame}
    
    \begin{frame}{LoRA: Algorithm}
    	\begin{algorithmic}
    		\LComment{====Initialization====} \ForAll{$W \in all\_weights$}
    		\State Set $\mathbf{W}$ to read-only
    		\Comment{Original knowledge preserved} \EndFor
    		
    		\ForAll{$W_u \in target\_modules$} \State Initialize $\mathbf{A}$
    		randomly \State Initialize $\mathbf{B} = \mathbf{0}$
    		\Comment{Contributes zero to the output initially} \EndFor
    		
    		\LComment{====Backpropagation====} \ForAll{$X \in input\_embeds$}
    		\State $\mathbf{\Delta W} = \frac{\alpha}{r} \mathbf{A} \mathbf{B}$
    		\State $\mathbf{H} = \mathbf{W} X + \mathbf{\Delta W} X$
    		\Comment{Output = original + adaptation} \EndFor
    		\LComment{====End of training====} \State
    		$\mathbf{W}' = \mathbf{W} + \frac{\alpha}{r} (\mathbf{A} \mathbf{B})$
    		\Comment{Merge original and adaptation}
    	\end{algorithmic}
    \end{frame}

    \begin{frame}
    {LoRA: Practical Considerations}
        \begin{itemize}
            \item How many ranks do I use?
            \begin{itemize}
                \item $r \in [1,16]$: fast, easy, recommended for minor
                adjustments. \item $r \in [16,64]$: recommended for most domain
                adaptation scenarios. \item $r \in [64, 256]$: may hit
                computational bottlenecks, only use for dramatic shifts in
                domain and knowledge.
            \end{itemize}

            \item How do I set $\alpha$?
            \begin{itemize}
                \item Usually set to a multiple of $r$ . \item Very low
                $\alpha \rightarrow$ minimal contribution. Very high
                $\alpha \rightarrow$ destruction of model knowledge. \item
                $\alpha = 2r$ usually used as a starting point.
            \end{itemize}
        \end{itemize}
    \end{frame}

    \begin{frame}
    {LoRA: Practical Considerations}
        \begin{itemize}
            \item What modules do I change? \item Start from the top of this
            list and go down as required:
            \begin{enumerate}
                \item
                \textbf{Query ($\text{q\_proj}$) and Value ($\text{v\_proj}$):}
                Most frequently adapted; control \alert{how the model attends}
                to input. \item \textbf{Key ($\text{k\_proj}$):} Increases
                \alert{information retrieval} capacity. \item
                \textbf{Output ($\text{o\_proj}$):} Less common, but useful for
                modifying \alert{integrated} attended information. \item
                \textbf{Feed-forward Networks (MLP):} Possible to adapt, but
                generally offer less parameter benefit compared to attention
                layers.
            \end{enumerate}
        \end{itemize}
    \end{frame}

    \begin{frame}
    {LoRA: Practical Considerations}
        \begin{itemize}
            \item Should we merge the LoRA weights?
        \end{itemize}

        \begin{center}
            $\mathbf{W}' = \mathbf{W} + \frac{\alpha}{r} (\mathbf{A} \mathbf{B})$
        \end{center}

        \begin{columns}
            \begin{column}{0.45\columnwidth}
                \begin{block}{Yes}
                    \begin{itemize}
                        \item Lowers \alert{GPU memory} and execution
                        \alert{time}.
                        \begin{itemize}
                            \item More significant if $r$ is large.
                        \end{itemize}
                        \item Deployment may be easier.
                    \end{itemize}
                \end{block}
            \end{column}

            \begin{column}{0.45\columnwidth}
                \begin{block}{No}
                    \begin{itemize}
                        \item You can use the same model with \alert{many}
                        finetuned adapters which are
                        \alert{dynamically selected} depending on your
                        task.\note{Mention storage requirements}
                    \end{itemize}
                \end{block}
            \end{column}
        \end{columns}
    \end{frame}

    \begin{frame}{QLoRA: Quantization}
        \begin{columns}
            \begin{column}{0.45\textwidth}
                \begin{itemize}
                    \item Quantization refers to dropping the size (not number)
                    of the LLM parameters.

                    \item Normally, each parameter is stored in a 16 bit
                    floating point.

                    \item Quantization can drop them to usually 4 bits,
                    resulting in \alert{$\times 4$ less} GPU memory and
                    inference costs.
                \end{itemize}
            \end{column}

            \begin{column}{0.45\textwidth}

    \end{frame}

    \begin{frame}{QLoRA: Concept}
        \begin{columns}
            \begin{column}{0.45\textwidth}
                \begin{itemize}
                    \item Load a 4-bit quantized model using a normal
                    approximation of weights.

                    \item Run LoRA in full precision, \alert{dequantizing} the
                    model weights \alert{one at a time}.

                    \item Continue as normal.
                \end{itemize}
            \end{column}

            \begin{column}{0.45\textwidth}
                \begin{figure}
                    \includegraphics[width=\columnwidth]{qlora.drawio.png}
                    \centering
                \end{figure}
            \end{column}
        \end{columns}
    \end{frame}

    \begin{frame}{QLoRA: Algorithm}
        \begin{algorithmic}
            \LComment{====Initialization====} \LComment{Same as before}

            \LComment{====Backpropagation====} \ForAll{$X \in input\_embeds$}
            \State $\mathbf{\Delta W} = \frac{\alpha}{r} \mathbf{A} \mathbf{B}$
            \State $\mathbf{H} =$ \alert{\Call{dequant}{$\mathbf{W}$}}
            $X + \mathbf{\Delta W} X$ \Comment{Output = original + adaptation}
            \EndFor \LComment{====End of training====}
            \LComment {Only attempt dynamic merge. Ignore this advice at your own peril, or if you really know what you are doing.}
        \end{algorithmic}
    \end{frame}

    \begin{frame}{QLoRA: Pros and cons}
        \begin{columns}
            \begin{column}{0.4\columnwidth}
                \begin{block}{Pros}
                    \begin{itemize}
                        \item Frozen weights no longer monopolize
                        \alert{GPU memory} and inference \alert{time}. \item
                        Allows training in \alert{consumer-grade} hardware.
                        \item Massively \alert{reduces training} times.
                    \end{itemize}
                \end{block}
            \end{column}

            \begin{column}{0.4\columnwidth}
                \begin{block}{Cons}
                    \begin{itemize}
                        \item End-result is a quantized $\rightarrow$
                        \alert{
                            less
                            powerful
                        }
                        model. \item Dequantization imposes a $\approx 10\%$ tax
                        on training times. \item Merging is now much messier.
                    \end{itemize}
                \end{block}
            \end{column}
        \end{columns}
    \end{frame}

    \begin{frame}{Summing up}

        \begin{itemize}
            \item Full finetuning for LLMs is out of the question. \item
            However, weight updates are usually \alert{low-rank}. \item Thus, we
            can \alert{approximate} full finetuning with LoRA. \item We may use
            QLoRA instead to make it much more \alert{scalable}.
        \end{itemize}
    \end{frame}
